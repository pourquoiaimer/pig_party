/* ===== FRUIT 邏輯 BEGIN（純函式：禁 DOM / blip / performance / Math.random）=====
   搶水果：左右移動接掉落物，45 秒或先到 15 分。RNG 由呼叫端注入（伺服器負責洗）。
   驗證：tests/fruit.logic.test.mjs
   ===== */
const FRG = { W: 960, GY: 540 - 74, WIN: 15, TIME: 45, SPD: 380, BOOST: 1.8, WIDE: 1.6 };

function frCreate() {
  return {
    P: [ { x: FRG.W * 0.25, boost: 0, wide: 0, sc: 0 },
         { x: FRG.W * 0.75, boost: 0, wide: 0, sc: 0 } ],
    items: [],                 // { x, y, vy, kind } kind 0=水果 1=boost 2=wide
    time: FRG.TIME, spawnT: 0, propT: 4,
    over: false, winner: -1, seed: 1,
  };
}

/** 一步物理（dt 秒）。rng 由外部給（同一 seed 同一序列） */
function frStep(S, dt, inp, rng) {
  const ev = [];
  if (S.over) return ev;
  S.time -= dt;
  S.spawnT -= dt; S.propT -= dt;
  if (S.spawnT <= 0) {
    S.items.push({ x: 40 + rng() * (FRG.W - 80), y: -30, vy: 210 + rng() * 150, kind: 0, emo: (rng() * 5) | 0 });
    S.spawnT = 0.6 - Math.min(0.3, (FRG.TIME - S.time) * 0.008);
  }
  if (S.propT <= 0) {
    S.items.push({ x: 40 + rng() * (FRG.W - 80), y: -30, vy: 170, kind: rng() < 0.5 ? 1 : 2, emo: 0 });
    S.propT = 5 + rng() * 4;
  }
  S.P.forEach((p, pi) => {
    const hh = inp[pi] || { l: 0, r: 0 };
    p.x += ((hh.r ? 1 : 0) - (hh.l ? 1 : 0)) * FRG.SPD * (p.boost > 0 ? FRG.BOOST : 1) * dt;
    p.x = Math.max(pi === 0 ? 20 : FRG.W / 2 + 10, Math.min(pi === 0 ? FRG.W / 2 - 10 : FRG.W - 20, p.x));
    if (p.boost > 0) p.boost -= dt;
    if (p.wide > 0) p.wide -= dt;
  });
  for (let i = S.items.length - 1; i >= 0; i--) {
    const it = S.items[i];
    it.y += it.vy * dt;
    if (it.y < FRG.GY - 30) continue;
    let caught = false;
    for (let pi = 0; pi < 2 && !caught; pi++) {
      const p = S.P[pi], w = 88 * (p.wide > 0 ? FRG.WIDE : 1);
      if (Math.abs(it.x - p.x) < w / 2 && it.y < FRG.GY + 10) {
        if (it.kind === 0) { p.sc++; ev.push({ t: 'eat', p: pi, k: 'fruit' }); }
        else if (it.kind === 1) { p.boost = 4; ev.push({ t: 'eat', p: pi, k: 'boost' }); }
        else { p.wide = 5; ev.push({ t: 'eat', p: pi, k: 'wide' }); }
        caught = true;
      }
    }
    if (caught || it.y > FRG.GY + 44) S.items.splice(i, 1);
  }
  if (S.P[0].sc >= FRG.WIN || S.P[1].sc >= FRG.WIN) {
    S.over = true; S.winner = S.P[0].sc >= FRG.WIN ? 0 : 1; ev.push({ t: 'win', p: S.winner });
  } else if (S.time <= 0) {
    S.time = 0; S.over = true;
    S.winner = S.P[0].sc === S.P[1].sc ? -1 : (S.P[0].sc > S.P[1].sc ? 0 : 1);
    ev.push({ t: 'win', p: S.winner });
  }
  return ev;
}


  /* ---------- 09E 連線模式：插值 + 由伺服器快照重建畫面 ---------- */
  let fPrev = null, fCur = null, fPrevT = 0, fCurT = 0;
  const animY = new Map();                       // (kind:x) → 目前畫的 y（用來平滑掉落）
  function netRender(dt) {
    const fr = NETGAME.fruit;
    if (fr && fr.S && fr.appliedK !== fr.k) {
      fr.appliedK = fr.k; fPrev = fCur; fPrevT = fCurT; fCur = fr.S; fCurT = performance.now();
    }
    if (!fCur) return;
    let a = 1;
    if (fPrev && fCurT > fPrevT) a = Math.min(1, Math.max(0, (performance.now() - fCurT + 34) / (fCurT - fPrevT)));
    for (let i = 0; i < 2; i++) {
      const A = fPrev ? fPrev.px[i] : fCur.px[i], B = fCur.px[i];
      P[i].x = Math.abs(B - A) > 120 ? B : A + (B - A) * a;
      P[i].sc = fCur.sc[i] | 0;
    }
    time = +fCur.tm || 0;
    const next = [], seen = new Set();
    for (const row of (fCur.it || [])) {
      const ix = row[0], iy = row[1], kind = row[2] | 0;
      const key = kind + ':' + Math.round(ix);
      seen.add(key);
      let y = animY.has(key) ? animY.get(key) : iy;
      y += (iy - y) * Math.min(1, dt * 18);
      animY.set(key, y);
      next.push({ x: ix, y, vy: 0, k: kind === 1 ? 'boost' : kind === 2 ? 'wide' : 'fruit',
                  emo: kind === 1 ? '👟' : kind === 2 ? '🧺' : EM[Math.abs(Math.round(ix)) % EM.length] });
    }
    for (const key of [...animY.keys()]) if (!seen.has(key)) animY.delete(key);
    items.length = 0;
    for (const it of next) items.push(it);
    $('s1').textContent = P[0].sc; $('s2').textContent = P[1].sc;
    $('st1').textContent = ''; $('st2').textContent = '';
    over = !!fCur.ov; winner = fCur.win | 0;
    draw();
  }



/** 收到搶水果快照 */
function netFruitOnState(m) {
  if (!NETGAME.active || NETGAME.game !== 'fruit') return;
  const s = m && m.s;
  if (!s || !Array.isArray(s.px) || s.px.length !== 2) return;
  if (typeof m.k === 'number') { if (m.k <= NETGAME.fruit.k) return; NETGAME.fruit.k = m.k; }
  NETGAME.fruit.S = s;
  for (const e of (Array.isArray(m.ev) ? m.ev : [])) {
    if (e.t !== 'eat') continue;
    blip(e.k === 'boost' ? 900 : e.k === 'wide' ? 760 : 620, 0.07, 0.06);
  }
}
