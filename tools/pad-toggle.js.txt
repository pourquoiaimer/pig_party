/* ── (3) JS：setupTouch(); 之後 ───────────────────────────── */
const PAD_LS = 'pigParty.pad.v1';
let padSaved = 'on';
try { padSaved = localStorage.getItem(PAD_LS) || 'on'; } catch (e) {}

function padOn() { return !document.body.classList.contains('pad-off'); }
function applyPadState() {
  document.body.classList.toggle('pad-off', padSaved === 'off');
  const b = $('padToggle');
  if (b) {
    b.textContent = padOn() ? '⌨ 隱藏按鍵' : '🎮 顯示按鍵';
    b.setAttribute('aria-pressed', padOn() ? 'true' : 'false');
  }
}
(function initPadToggle() {
  const b = $('padToggle');
  if (!b) return;
  applyPadState();
  b.addEventListener('click', (e) => {
    e.preventDefault(); e.stopPropagation();
    padSaved = padOn() ? 'off' : 'on';
    try { localStorage.setItem(PAD_LS, padSaved); } catch (err) {}
    applyPadState();
    blip(padOn() ? 520 : 320, 0.06, 0.05);
    b.blur();
  });
})();
